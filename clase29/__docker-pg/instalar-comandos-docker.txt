# Ejecutar estos comandos desde la carpeta clase29
# Crear la imagen usando el Dockerfile ubicado en __docker-pg
docker build -t imagen_admin -f .\__docker-pg\Dockerfile .\__docker-pg

# Instanciar un contenedor a partir de la imagen
docker run --name contenedor_admin -e POSTGRES_USER=root -e POSTGRES_PASSWORD=pass -e POSTGRES_DB=admin -p 5433:5432 -d imagen_admin

# Verificar la creación de la base de datos y la tabla

# Entrar al contenedor
docker exec -it contenedor_admin bash

# Dentro del contenedor ejecutar:
psql -U root -d admin --password
# Indicar la clave: pass

# Ejecutar consulta
SELECT * FROM usuarios;
# Debemos ver los usuarios insertados.
